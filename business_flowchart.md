    System[Портал "Нормоконтроль"]

        B[Выбор правил проверки]
        C[Постановка в очередь на обработку]
    end

    subgraph "Шаг 2: Автоматическая проверка"
        D[Чтение и анализ структуры документа]

        subgraph "Анализ текста"

        end

    D --> E
    E --> F & G
    F & G --> H
    H -- "Удаление неточных правок" --> I
    I --> J
    J --> K
